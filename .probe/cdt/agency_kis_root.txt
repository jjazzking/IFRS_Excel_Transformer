	var mainPopupSlides = [
		{
	        index: 1,
	        thumbnail: "/common/images/common/img_popup_slide_03.png",
	        popupTarget: "layerpop_notice1",
	        alt: "제39기 기업신용평가분석과정"

	var mainPopupSlideIndex = 0; // 배열 기준 1번 이미지 기본 노출
	
	/* HP-2608-062
	 * 메인 팝업 슬라이드 초기화
	 * 디지털플랫폼 이미지를 기본 이미지로 노출
	 */
	function fnInitMainPopupSlider() {
		var $paging = $("#main_popup_slider_paging");

	    $paging.empty();

	    $.each(mainPopupSlides, function (arrayIndex, slide) {
	        var $button = $("<button>", {
	            type: "button",
	            class: "main_popup_slide_number",
	            text: slide.index,
	            "data-slide-index": arrayIndex,
	            "aria-label": slide.index + "번 슬라이드 보기",
	            role: "tab"
	        });

	        $paging.append($button);
	    });

	    $(document)
		    .off("click.mainPopupSlider", ".main_popup_slide_number")
		    .on("click.mainPopupSlider", ".main_popup_slide_number", function () {
		        var selectedIndex = Number($(this).attr("data-slide-index"));
		        fnChangeMainPopupSlide(selectedIndex, true);   // true = 애니메이션 없이 즉시 전환
		        fnStartMainPopupAutoPlay();                    // 타이머 리셋
	    });

	    $(document)
	        .off("click.mainPopupImage", "#main_popup_slide_link")
	        .on(
	            "click.mainPopupImage",
	            "#main_popup_slide_link",
	            function () {
	                fnOpenMainPopupSlide();
	            }
	        );

	    fnChangeMainPopupSlide(mainPopupSlideIndex);
	    
	 	// 자동 넘김 시작
    	fnStartMainPopupAutoPlay();
    	
	 	// 마우스 올리면 정지, 벗어나면 재시작
    	$("#main_popup_slider")
    		.off("mouseenter.mps mouseleave.mps")
    		.on("mouseenter.mps", function(){ fnStopMainPopupAutoPlay(); })
    		.on("mouseleave.mps", function(){ fnStartMainPopupAutoPlay(); });
	}

	var mainPopupAnimating = false;   // 애니메이션 중복 방지

	function fnChangeMainPopupSlide(selectedIndex, skipAnim){
	    var slide = mainPopupSlides[selectedIndex];
	    if(!slide){ return; }
	    if(mainPopupAnimating){ return; }              // 진행 중이면 무시

	    var $view    = $(".main_popup_slider_view");
	    var $track   = $view.find(".mps_track");
	    var $current = $("#main_popup_slide_image");

	    // 번호 active 상태 갱신
	    $(".main_popup_slide_number").removeClass("active").attr("aria-selected","false");
	    $(".main_popup_slide_number").eq(selectedIndex).addClass("active").attr("aria-selected","true");

	    // 첫 로드(현재 이미지 src 없음) 또는 애니메이션 스킵이면 즉시 표시
	    if(!$current.attr("src") || skipAnim){
	        $current.attr("class", "mps_current").css("left", "0")
	                .attr("src", slide.thumbnail).attr("alt", slide.alt);
	        mainPopupSlideIndex = selectedIndex;
	        return;
	    }

	    mainPopupAnimating = true;

	    // 오른쪽 밖에 새 이미지 배치
	    var $incoming = $("<img>", {
	        "class": "mps_incoming",
	        src: slide.thumbnail,
	        alt: slide.alt
	    });
	    $track.append($incoming);

	    // reflow 강제 후 애니메이션 시작
	    $view[0].offsetWidth;
	    $view.addClass("mps_animating");

	    // 애니메이션 종료(0.5s) 후 정리
	    setTimeout(function(){
	        $current.attr("src", slide.thumbnail).attr("alt", slide.alt);
	        $current.attr("class", "mps_current").css("left", "0");
	        $incoming.remove();
	        $view.removeClass("mps_animating");

	        mainPopupSlideIndex = selectedIndex;
	        mainPopupAnimating = false;
	    }, 500);
	}
	
	/**
	 * 현재 선택된 슬라이드의 팝업을 fnBannerPop3으로 띄운다.
	 */
	function fnOpenMainPopupSlide() {
	    var slide = mainPopupSlides[mainPopupSlideIndex];

	    if (!slide || !slide.popupTarget) {
	        return;
	    }

	    fnStopMainPopupAutoPlay();   // 팝업 보는 동안 자동 넘김 정지

	    // 이미 떠 있는 동일 팝업 제거 (중복 append 방지)
	    $("#" + slide.popupTarget).remove();

	    fnBannerPop3(slide.popupTarget);
	}

	/**
	 * 자동 넘김 시작 (기존 타이머는 정리 후 재시작)
	 */
	function fnStartMainPopupAutoPlay(){
	    fnStopMainPopupAutoPlay();
	    if(mainPopupSlides.length <= 1){ return; }   // 1개면 불필요
	    mainPopupSlideTimer = setInterval(fnNextMainPopupSlide, MAIN_POPUP_SLIDE_INTERVAL);
	}

	/**
	 * 자동 넘김 정지
	 */
	function fnStopMainPopupAutoPlay(){
	    if(mainPopupSlideTimer){
	        clearInterval(mainPopupSlideTimer);
	        mainPopupSlideTimer = null;
	    }
	}

	/**
	 * 다음 슬라이드로 이동 (마지막이면 처음으로 순환)
	 */
	function fnNextMainPopupSlide(){
	    var nextIndex = (mainPopupSlideIndex + 1) % mainPopupSlides.length;
	    fnChangeMainPopupSlide(nextIndex);
	}

            url: "/getMyPortpolioFolder.json",
            type: 'post',
            contentType :"application/json",
            data : JSON.stringify({"flag" : gMyPortpolioFlag}),
            success: function(data, status, xhr) {
            	if(data.loginFail=="Y"){
            		alert("로그인이 필요한 기능입니다");
            		closePopUpMyPortpolio();
            		return;
            	}
            	
            	//console.log("success",data);
            	
           		for(var i=0;i<data.folderList.length;i++){
  
           			$("#mypfUl").append(
           					"<li><a href=\"javascript:myPFolderSelect('" + data.folderList[i].idx + "')\">" + data.folderList[i].folderName + "</a></li>"
					);
           		}            	 
            }
	 		,error: function(data, status, xhr) {
	 			if(data.status == '599'){
  					if(lang == "en"){
  						alert('Login limit exceeded.'); 
  						location.href='/en/';
  					}else{
  						alert('중복으로 로그인 되었거나 오랫동안 사용하지 않아서 로그아웃 되었습니다.');
  						location.href='/';
  					}
  				}
	 			//console.log("error",xhr);
	 		}
       });        
    }
